// cursor-trail.js

// Создаём элегантный след в стиле дизайн-портфолио
const emojis = ['✨', '⭐', '💫', '✦'];
const randomEmoji = emojis[Math.floor(Math.random() * emojis.length)];

document.addEventListener('mousemove', (e) => {
    // Создаём элемент следа
    const trail = document.createElement('div');
    
    // Стилизация под ваш бренд (мягкие тона)
trail.textContent = randomEmoji;
trail.style.cssText = `
    position: fixed;
    left: ${e.clientX - 4}px;
    top: ${e.clientY - 4}px;
    font-size: 128px;
    pointer-events: none;
    z-index: 9999;
    transition: all 0.6s ease;
    opacity: 0.6;
    transform: scale(0.5) rotate(0deg);
`;
    document.body.appendChild(trail);
    
// Анимация
requestAnimationFrame(() => {
    trail.style.transform = 'scale(1) rotate(20deg)';
    trail.style.opacity = '0.8';
});

setTimeout(() => {
    trail.style.opacity = '0';
    trail.style.transform = 'scale(0.3) rotate(40deg) translateY(-20px)';
    setTimeout(() => trail.remove(), 600);
}, 250);
    
    // Ограничиваем количество следов
    trailCount++;
    if (trailCount > MAX_TRAILS) {
        const oldTrail = document.body.querySelector('.trail-effect');
        if (oldTrail) oldTrail.remove();
        trailCount = MAX_TRAILS;
    }
});

console.log('✅ Cursor trail эффект загружен! Дизайн-портфолио Даши');
