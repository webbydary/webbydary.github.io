// 📸 СПИСОК ВАШИХ СТИКЕРОВ (укажите правильные имена файлов)
const stickers = [
    'images/stickers/sticker01.png',
    'images/stickers/sticker02.png',
    'images/stickers/sticker03.png',
    'images/stickers/sticker04.png',
    'images/stickers/sticker05.png',
    'images/stickers/sticker06.png',
    'images/stickers/sticker07.png',
    'images/stickers/sticker08.png',
    'images/stickers/sticker09.png',
    'images/stickers/sticker10.png',
    'images/stickers/sticker11.png',
    'images/stickers/sticker12.png',
    'images/stickers/sticker13.png',
    'images/stickers/sticker14.png',
    'images/stickers/sticker15.png',
    'images/stickers/sticker16.png',
    'images/stickers/sticker17.png',
    'images/stickers/sticker18.png',
    // Добавьте столько, сколько у вас есть
];

// Заранее загружаем все изображения для быстрой работы
const loadedImages = [];
stickers.forEach((src) => {
    const img = new Image();
    img.src = src;
    loadedImages.push(img);
});

let trailCount = 0;
const MAX_TRAILS = 4; // Максимальное количество стикеров на экране

document.addEventListener('mousemove', (e) => {
    // Выбираем случайный стикер
    const randomIndex = Math.floor(Math.random() * loadedImages.length);
    const selectedImage = loadedImages[randomIndex];
    
    // Создаём элемент с изображением
    const trail = document.createElement('img');
    trail.src = selectedImage.src;
    
    // Случайный размер (от 30px до 60px)
    const size = 150 + Math.random() * 190;
    
    // Стилизация
    trail.style.cssText = `
        position: fixed;
        left: ${e.clientX - size/2}px;
        top: ${e.clientY - size/2}px;
        width: ${size}px;
        height: ${size}px;
        pointer-events: none;
        z-index: 9999;
        opacity: 0;
        transform: scale(0.3) rotate(${Math.random() * 360}deg);
        transition: all 0.6s cubic-bezier(0.34, 1.56, 0.64, 1);
        object-fit: contain;
        filter: drop-shadow(0 4px 8px rgba(0,0,0,0.1));
    `;
    
    document.body.appendChild(trail);
    
    // Анимация появления
    requestAnimationFrame(() => {
        trail.style.opacity = '0.9';
        trail.style.transform = `scale(1) rotate(${Math.random() * 20 - 10}deg)`;
    });
    
    // Удаляем с затуханием
    setTimeout(() => {
        trail.style.opacity = '0';
        trail.style.transform = `scale(0.5) rotate(${Math.random() * 40 - 20}deg) translateY(-30px)`;
        setTimeout(() => trail.remove(), 700);
    }, 300);
    
    // Ограничиваем количество стикеров на экране
    trailCount++;
    if (trailCount > MAX_TRAILS) {
        const oldTrail = document.querySelector('.sticker-trail');
        if (oldTrail) {
            oldTrail.style.opacity = '0';
            setTimeout(() => oldTrail.remove(), 300);
        }
        trailCount = MAX_TRAILS;
    }
});

console.log('✅ Стикеры загружены! Количество:', loadedImages.length);
