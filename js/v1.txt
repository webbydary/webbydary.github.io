// cursor-trail.js

document.addEventListener('mousemove', (e) => {
    // Создаём элемент следа
    const trail = document.createElement('div');
    const size = 8 + Math.random() * 10; // разный размер
    
    // Стилизация под ваш бренд (мягкие тона)
trail.style.cssText = `
    position: fixed;
    left: ${e.clientX - size/2}px;
    top: ${e.clientY - size/2}px;
    width: ${size}px;
    height: ${size}px;
    background: rgba(203, 202, 204, ${0.3 + Math.random() * 0.3});
    border-radius: 50%;
    pointer-events: none;
    z-index: 9999;
    transition: all 0.4s ease-out;
    opacity: 0.7;
`;
    document.body.appendChild(trail);
    
// Анимация
requestAnimationFrame(() => {
    trail.style.transform = 'scale(1) rotate(20deg)';
    trail.style.opacity = '0.8';
});

setTimeout(() => {
    trail.style.opacity = '0';
    trail.style.transform = 'scale(2)';
    setTimeout(() => trail.remove(), 400);
}, 100);
    
    // Ограничиваем количество следов
    trailCount++;
    if (trailCount > MAX_TRAILS) {
        const oldTrail = document.body.querySelector('.trail-effect');
        if (oldTrail) oldTrail.remove();
        trailCount = MAX_TRAILS;
    }
});

console.log('✅ Cursor trail эффект загружен! Дизайн-портфолио Даши');
